```python
"""Script de verificação da Fase 5."""
import sys
import os
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from backend.db import DB_PATH, get_connection
from backend.irai.engine import IRAIEngine

def main():
    print("=== Verificação da Fase 5: Expansão de Ativos ===")
    engine = IRAIEngine(db_path=DB_PATH)
    targets = [t["target"] for t in engine.registered_targets]
    print(f"Total de alvos no Engine: {len(targets)}")
    
    new_assets = ["DE40", "US2000", "JP225", "CHINA50"]
    for asset in new_assets:
        found = any(t["target"] == asset for t in engine.registered_targets)
        status = "OK" if found else "FALHA"
        print(f"  Asset {asset:10s}: {status}")

    assert all(any(t["target"] == asset for t in engine.registered_targets) for asset in new_assets), "Nem todos os novos ativos foram carregados no engine!"
    print("\n[SUCESSO] Todos os 4 novos índices globais foram integrados com sucesso no Rastro IRAI!")

if __name__ == "__main__":
    main()
```
